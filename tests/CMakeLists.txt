find_package(GTest QUIET)
if(NOT GTest_FOUND)
  include(FetchContent)
  FetchContent_Declare(googletest
    URL https://github.com/google/googletest/archive/refs/tags/v1.14.0.zip)
  set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
  FetchContent_MakeAvailable(googletest)
endif()
include(GoogleTest)

file(GLOB TEST_SRC CONFIGURE_DEPENDS *.cpp)
add_executable(lms_tests ${TEST_SRC})
target_link_libraries(lms_tests PRIVATE lms_presentation lms_infrastructure GTest::gtest_main)
target_include_directories(lms_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
gtest_discover_tests(lms_tests DISCOVERY_TIMEOUT 60)
